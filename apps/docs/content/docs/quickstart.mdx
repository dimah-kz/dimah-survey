This guide publishes a SurveyJS document, starts a response, renders its frozen
definition, saves page progress, and submits it.

<Callout type="warn">
  This quickstart uses process-local memory and a fixed respondent. Before
  production, use a durable store, derive identity from the session, and guard
  the editor route.
</Callout>

<Steps>

<Step>

### Install

```npm
npm i @dimah-survey/server @dimah-survey/react survey-core survey-react-ui
```

`survey-react-ui` renders questions. dimah-survey owns the lifecycle around the
SurveyJS document.

</Step>

<Step>

### Create the server

Create a fill instance and an editor instance on one store.

```ts title="lib/survey.ts"
import {
  dimahSurvey,
  guardRespondent,
  memoryAdapter,
} from "@dimah-survey/server";

const database = memoryAdapter();

export const editor = dimahSurvey({
  audience: "editor",
  database,
});

export const fill = dimahSurvey({
  audience: "fill",
  database,
  guard: guardRespondent("demo"),
});
```

The guard stamps `"demo"` on every response.

</Step>

<Step>

### Publish a SurveyJS document

Saving writes `draftJson`. Publishing copies that draft to `publishedJson`.

```ts title="lib/welcome.ts"
import { isAPIError } from "@dimah-survey/server";
import { editor } from "./survey";

const definition = {
  title: "Welcome",
  pages: [
    {
      name: "welcome",
      elements: [
        { type: "text", name: "name", title: "Your name", isRequired: true },
        {
          type: "rating",
          name: "score",
          title: "How was it?",
          isRequired: true,
        },
      ],
    },
  ],
};

export async function ensureWelcomeSurvey() {
  try {
    await editor.api.getSurvey({ query: { id: "welcome" } });
  } catch (error) {
    if (!isAPIError(error) || error.code !== "SURVEY_NOT_FOUND") throw error;

    await editor.api.saveSurvey({
      body: { id: "welcome", slug: "welcome", draftJson: definition },
    });
    await editor.api.publishSurvey({ body: { id: "welcome" } });
  }
}
```

`getSurvey` throws `SURVEY_NOT_FOUND` when the row is missing.

</Step>

<Step>

### Mount both audiences

Each audience gets its own route. The handler serves only that audience.

<Tabs items={["Next.js", "Express", "Hono", "Fastify", "SvelteKit"]}>
  <Tab value="Next.js">

```ts title="app/api/survey/[[...path]]/route.ts"
import { toNextJsHandler } from "@dimah-survey/server/next";
import { fill } from "@/lib/survey";

export const { GET, POST, PUT, PATCH, DELETE } = toNextJsHandler(fill);
```

```ts title="app/api/admin/survey/[[...path]]/route.ts"
import { toNextJsHandler } from "@dimah-survey/server/next";
import { editor } from "@/lib/survey";

export const { GET, POST, PUT, PATCH, DELETE } = toNextJsHandler(editor);
```

  </Tab>
  <Tab value="Express">

```ts title="server.ts"
import express from "express";
import { toExpressHandler } from "@dimah-survey/server/express";
import { editor, fill } from "./survey";

const app = express();

app.all("/api/survey/*", toExpressHandler(fill));
app.all("/api/admin/survey/*", toExpressHandler(editor));
app.use(express.json());
```

Register these routes before `express.json()` so the adapter can read the body.

  </Tab>
  <Tab value="Hono">

```ts title="server.ts"
import { Hono } from "hono";
import { toHonoHandler } from "@dimah-survey/server/hono";
import { editor, fill } from "./survey";

const app = new Hono();
const methods = ["GET", "POST", "PUT", "PATCH", "DELETE"] as const;

app.on(methods, "/api/survey/*", toHonoHandler(fill));
app.on(methods, "/api/admin/survey/*", toHonoHandler(editor));
```

  </Tab>
  <Tab value="Fastify">

```ts title="server.ts"
import Fastify from "fastify";
import { toFastifyHandler } from "@dimah-survey/server/fastify";
import { editor, fill } from "./survey";

const app = Fastify();

app.all("/api/survey/*", toFastifyHandler(fill));

```

Leave the JSON parser off these paths so the adapter can read the body.

  </Tab>
  <Tab value="SvelteKit">


  </Tab>
</Tabs>

Fill is `/api/survey`. Editor is `/api/admin/survey`. Set `basePath` when the
route differs, and pass that same path as the browser client's `baseURL`.
Elysia and Node are on [Mount the server](/docs/integration).

</Step>

<Step>

### Create the browser client

The fill view below is React. `createFillClient()` calls `/api/survey`.

```ts title="lib/fill-client.ts"
"use client";

import { createFillClient } from "@dimah-survey/react";

export const fillClient = createFillClient();
```

Browser methods take a flat object. `fill.api` takes `{ body }` or `{ query }`.

</Step>

<Step>

### Render the response snapshot

```tsx title="components/fill.tsx"
"use client";

import dynamic from "next/dynamic";
import { useSurveyResponse } from "@dimah-survey/react";
import { fillClient } from "@/lib/fill-client";

import "survey-core/survey-core.min.css";

const Survey = dynamic(
  () => import("survey-react-ui").then((mod) => mod.Survey),
  { ssr: false },
);

export function Fill({ responseId }: { responseId: string }) {
  const { model, error, saveError, stale, reload } = useSurveyResponse({
    client: fillClient,
    responseId,
  });

  if (error) return <p>{error.message}</p>;
  if (!model) return null;

  return (
    <>
      {saveError ? <p>{saveError.message}</p> : null}
      {stale ? <button onClick={reload}>Reload</button> : null}
      <Survey model={model} />
    </>
  );
}
```

The hook builds a `Model` from `response.definition` and saves the full
`survey.data` on the next page. It does not render questions. Next.js loads
`<Survey>` in the browser; other React apps can import it directly.

</Step>

<Step>

### Start the response

```tsx title="app/page.tsx"
import { Fill } from "@/components/fill";
import { fill } from "@/lib/survey";
import { ensureWelcomeSurvey } from "@/lib/welcome";

export default async function Page() {
  await ensureWelcomeSurvey();
  const response = await fill.api.startResponse({
    body: { surveyId: "welcome" },
  });

  return <Fill responseId={response.id} />;
}
```

This page is Next.js. Any server runtime calls `fill.api.startResponse` the
same way. The first call copies the published document onto the response.
Refreshing resumes that draft. A later publish leaves it unchanged.

</Step>

</Steps>

## Make it production-ready

- Replace `memoryAdapter()` with the SQL store or your own durable
  `SurveyStore`. See [Persistence](/docs/persistence).
- Read the respondent from your session and protect the editor audience. See
  [Security](/docs/security).
- Add Creator autosave without coupling it to publish. See
  [Survey Creator](/docs/creator).
