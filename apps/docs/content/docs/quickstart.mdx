This guide uses Next.js. The same server API works on every supported runtime.

### Install [step]

```npm
npm i @dimah-survey/server @dimah-survey/db @dimah-survey/react react react-dom survey-core survey-react-ui
```

`survey-react-ui` renders the questions.

### Add the tables [step]

Add `dimah_survey` and `dimah_response` to the database you already use, migrate,
and export `database`.

### Create the server [step]

Create a fill instance and an editor instance. Replace `getSession` with your
authentication code. The browser never chooses the respondent id.

```ts title="lib/survey.ts"
import {
  APIError,
  SURVEY_ERROR_CODES,
  dimahSurvey,
  guardRespondent,
} from "@dimah-survey/server";

import { database } from "@/db";

export const editor = dimahSurvey({
  audience: "editor",
  database,
  guard: async ({ request }) => {
    const session = request ? await getSession(request) : null;
    if (session?.user.role !== "editor") {
      throw APIError.from("FORBIDDEN", SURVEY_ERROR_CODES.FORBIDDEN);
    }
  },
});

export const fill = dimahSurvey({
  audience: "fill",
  database,
  guard: async (context) => {
    const session = context.request
      ? await getSession(context.request)
      : null;
    if (!session) {
      throw APIError.from("FORBIDDEN", SURVEY_ERROR_CODES.FORBIDDEN);
    }
    return guardRespondent(session.user.id)(context);
  },
});
```

Both instances share `database`. See [Authorization](/docs/security) for
anonymous responses.

### Publish a SurveyJS document [step]

Saving writes `draftJson`. Publishing stores that draft as the current version.
Seed one document when the server starts:

```ts title="lib/welcome.ts"
import { database } from "@/db";

const definition = {
  title: "Welcome",
  pages: [
    {
      name: "welcome",
      elements: [
        { type: "text", name: "name", title: "Your name", isRequired: true },
        {
          type: "rating",
          name: "score",
          title: "How was it?",
          isRequired: true,
        },
      ],
    },
  ],
};

export async function seedWelcomeSurvey() {
  if (await database.getSurvey("welcome")) return;

  await database.saveSurvey({
    id: "welcome",
    slug: "welcome",
    draftJson: definition,
  });
  await database.publishSurvey({ id: "welcome" });
}
```

```ts title="instrumentation.ts"
export async function register() {
  if (process.env.NEXT_RUNTIME !== "nodejs") return;
  const { seedWelcomeSurvey } = await import("@/lib/welcome");
  await seedWelcomeSurvey();
}
```

This seed calls the store directly. Authoring in the product should use the
editor API so publish hooks run.

### Mount both audiences [step]

Each audience has its own route. The handler serves only that audience.

```ts title="app/api/survey/[[...path]]/route.ts"
import { toNextJsHandler } from "@dimah-survey/server/next";

import { fill } from "@/lib/survey";

export const { GET, POST, PUT, PATCH, DELETE } = toNextJsHandler(fill);
```

```ts title="app/api/admin/survey/[[...path]]/route.ts"
import { toNextJsHandler } from "@dimah-survey/server/next";

import { editor } from "@/lib/survey";

export const { GET, POST, PUT, PATCH, DELETE } = toNextJsHandler(editor);
```

Fill is `/api/survey`. Editor is `/api/admin/survey`. Set `basePath` when the
route differs, and pass that same path as the browser client's `baseURL`.
Other runtimes are on [Mount the server](/docs/integration).

### Start the response [step]

```ts title="lib/start-response.ts"
import { headers } from "next/headers";

import { fill } from "@/lib/survey";

export async function startWelcomeResponse() {
  const response = await fill.api.startResponse({
    body: { surveyId: "welcome" },
    headers: await headers(),
  });
  return response.id;
}
```

Forward `headers()` so the fill guard sees the session. The first call stores
the current version on the response. A later call returns that respondent's open
draft.

### Render the response [step]

`createFillClient()` calls `/api/survey`. The hook reads the snapshot. `<Survey>` renders it.

```tsx title="components/fill.tsx"
"use client";

import dynamic from "next/dynamic";
import { createFillClient, useSurveyResponse } from "@dimah-survey/react";

import "survey-core/survey-core.min.css";

const Survey = dynamic(
  () => import("survey-react-ui").then((mod) => mod.Survey),
  { ssr: false },
);

const client = createFillClient();

export function Fill({ responseId }: { responseId: string }) {
  const { model, error, saveError, stale, reload } = useSurveyResponse({
    client,
    responseId,
  });

  if (error) return <p>{error.message}</p>;
  if (!model) return <p>Loading…</p>;

  return (
    <>
      {saveError ? <p>{saveError.message}</p> : null}
      {stale ? <button onClick={reload}>Reload</button> : null}
      <Survey model={model} />
    </>
  );
}
```

```tsx title="app/page.tsx"
import { Fill } from "@/components/fill";
import { startWelcomeResponse } from "@/lib/start-response";

export default async function Page() {
  const responseId = await startWelcomeResponse();
  return <Fill responseId={responseId} />;
}
```

The hook builds a `Model` from `response.definition` and saves the complete
`survey.data` object on the next page. It does not render SurveyJS.
