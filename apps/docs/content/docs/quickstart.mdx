### Create the server [step]

```ts title="lib/survey.ts"
import { dimahSurvey } from "@dimah-survey/server";

import { database } from "@/db";

export const editor = dimahSurvey({
  audience: "editor",
  database,
  guard: async () => {
    // Authorize the editor.
  },
});

export const fill = dimahSurvey({
  audience: "fill",
  database,
  guard: async () => {
    // Return { respondentId } or { anonymous: true }.
    return { anonymous: true };
  },
});
```

### Publish a survey [step]

Call `seedWelcomeSurvey()` when the process starts. Product authoring should use the editor API so publish hooks run.

```ts title="lib/welcome.ts"
import { database } from "@/db";

const definition = {
  title: "Welcome",
  pages: [
    {
      name: "welcome",
      elements: [
        { type: "text", name: "name", title: "Your name", isRequired: true },
        {
          type: "rating",
          name: "score",
          title: "How was it?",
          isRequired: true,
        },
      ],
    },
  ],
};

export async function seedWelcomeSurvey() {
  if (await database.getSurvey("welcome")) return;

  await database.saveSurvey({
    id: "welcome",
    slug: "welcome",
    draftJson: definition,
  });
  await database.publishSurvey({ id: "welcome" });
}
```

```ts
import { createServer } from "node:http";
import { toNodeHandler } from "@dimah-survey/server/node";
import { editor, fill } from "./survey";

const handleFill = toNodeHandler(fill);
const handleEditor = toNodeHandler(editor);

createServer((req, res) => {
  const url = req.url ?? "";
  if (url.startsWith("/api/admin/survey")) return handleEditor(req, res);
  if (url.startsWith("/api/survey")) return handleFill(req, res);
  res.statusCode = 404;
  res.end();
}).listen(3000);
```

### Start the response [step]

<Tabs items={["Next.js", "Hono", "Express"]}>
  <Tab value="Next.js">

```ts title="lib/start-response.ts"
import { headers } from "next/headers";

import { fill } from "@/lib/survey";

export async function startWelcomeResponse() {
  const response = await fill.api.startResponse({
    body: { surveyId: "welcome" },
    headers: await headers(),
  });
  return response.id;
}
```

### Render the response [step]

```tsx title="fill.tsx"
import { Survey } from "survey-react-ui";
import { createFillClient, useSurveyResponse } from "@dimah-survey/react";

import "survey-core/survey-core.min.css";

const client = createFillClient({ baseURL: "/api/survey" });

export function Fill({ responseId }: { responseId: string }) {
  const { model, error, saveError, stale, reload } = useSurveyResponse({
    client,
    responseId,
  });

  if (error) return <p>{error.message}</p>;
  if (!model) return <p>Loading…</p>;

  return (
    <>
      {saveError ? <p>{saveError.message}</p> : null}
      {stale ? <button onClick={reload}>Reload</button> : null}
      <Survey model={model} />
    </>
  );
}
```
